    {

    static double ApproachTime(FlightSettings flight)
    { return flight.FirstAction==FirstAction.Jump ? (350+Math.Sqrt(350*350-2*662.5*48))/662.5 : FlipTime(164); }
    static GameState Approach(FlightSettings flight,int stoneSide,double landingPlaneX,double clearX,double step)
    {
        var game=Prepare(LinkedFixture(stoneSide,clearX),flight,
            flight.FirstAction==FirstAction.Jump?stoneSide:1-stoneSide,step);
        MoveTo(game,landingPlaneX-220*ApproachTime(flight),step);Tap(game);return game;
    }
    static void AfterSemantics()

    static GameState FullRouteGame(Track track,FlightSettings flight,int kind,double step,out double[] times,
        int shiftIndex=-1,double shift=0,double[] baseline=null)
    {
        var route=kind==0?TracksFull.SafeRoute(track.Id,flight):kind==1?TracksFull.FullRoute(track.Id,flight):TracksFull.NegativeRoute(track.Id,flight);
        var game=Start(track,flight);times=new double[route.Length];
        for(int i=0;i<route.Length;i++)
        {
            if(i==shiftIndex)Advance(game,baseline[i]+shift-game.Elapsed,step);else MoveTo(game,route[i],step);
            times[i]=game.Elapsed;
            if(flight.AirRepeats==AirRepeatLimit.None)foreach(var p in game.Players)Check(p.Support!=SupportKind.None&&!p.Flipping,"Fast input uses real support input="+i);
            Tap(game);
        }
        Advance(game,100,step);
        double expected=kind==0?36:kind==1?33.2:33.5;
        foreach(var p in game.Players)
        {
            Check(p.FinishTime.HasValue&&p.Collisions==0,"real clean finish "+State(p));
            Near(p.Distance,7920,"physical finish line");Near(p.FinishTime.Value,expected,"canonical route result");
            Check(p.BoostsEarned==(kind==0?0:9)&&p.CompletedBoosts==p.BoostsEarned,"ordinary rewards fully served");
            Check(p.FinishBoostsEarned==(kind==0?0:1),"separate final reward");
            Check(p.SlowsEarned==(kind==2?1:0)&&p.CompletedSlows==p.SlowsEarned,"trap ledger fully served");
            Check(p.ActiveEffect==null&&p.PendingEffects.Count==0,"empty actual effect state");
            foreach(var zone in track.RiskZones)
                Check(!p.CollectedRiskIds.Contains(zone.Id)||p.SuccessfulRiskIds.Contains(zone.Id)||p.NegativeRiskIds.Contains(zone.Id)||p.FailedRiskIds.Contains(zone.Id),"no retained pending at finish "+zone.Id);
        }
        Near(game.Elapsed,expected,"actual game clock equals result, not only caption");
        return game;
    }
    static void FullAudit()
    {
        int windows=0;
        foreach(var flight in new[]{ExpectedMain,ExpectedFast})foreach(var original in TracksFull.All)
        {
            var track=TracksFull.Get(original.Id,flight);

            {
                context="full "+track.Id+" profile="+flight.FirstAction+" kind="+kind+" dt="+step;
                double[] ignored;FullRouteGame(track,flight,kind,step,out ignored);
            }
            context="full safe baseline "+track.Id+" "+flight.FirstAction;
            double[] times;FullRouteGame(track,flight,0,1d/144,out times);
            for(int index=0;index<times.Length;index++)foreach(double shift in new[]{-.06,.06})

                Near(shifted[index],times[index]+shift,"actual clock shift",2e-8);

                FinishOnlyDuel(track,flight,winner,step);
        }
        Console.WriteLine("PASS independent full routes: "+checks+" assertions / "+games+" games; safe/full/trap both profiles/three tracks/six dt, "+windows+" individual safe±60ms shifts and final-only paired duels.");
    }
    static void FinishOnlyDuel(Track track,FlightSettings flight,int winner,double step)

        context="final-only duel "+track.Id+" "+flight.FirstAction+" winner="+winner+" dt="+step;

// Source: Tools/TwoModesAudit/Program.cs
public static partial class TwoModesIndependentAudit
{
    static int checks, games;
    static string context;
    static readonly double[] Steps = { 1d/30, 1d/60, 1d/144, .317, 2.5, 100 };
    static readonly FlightSettings ExpectedFast = FlightPresets.FastFlip;
    static readonly FlightSettings ExpectedMain = FlightPresets.Chosen;
    const double R = 14, H = 240, Span = 212;
    static void Check(bool value, string message)
    { checks++; if (!value) throw new Exception(context + ": " + message); }
    static void Near(double actual, double expected, string message, double tolerance=5e-7)
    { Check(!double.IsNaN(actual) && Math.Abs(actual-expected)<=tolerance,
        message+" actual="+actual.ToString("R")+" expected="+expected.ToString("R")); }
    static double Wall(int side) { return side == 1 ? 226 : 14; }
    static double Away(int side) { return side == 1 ? -1 : 1; }
    static double FlipTime(double distance) { return (Math.Sqrt(35*35+2*2700*distance)-35)/2700; }
    static GameState Start(Track track, FlightSettings flight)
    {
        var game=RaceSimulation.CreateTrial(track,flight);
        Check(RaceSimulation.Press(game,0)&&RaceSimulation.Press(game,1),"two ready presses accepted");
        RaceSimulation.Advance(game,3); Check(game.Phase=="racing","actual countdown ends"); games++;
        return game;
    }
    static void Advance(GameState game,double seconds,double step)
    {
        Check(seconds>=-1e-9,"nonnegative advance request"); int guard=0;
        while(seconds>1e-13&&game.Phase=="racing")
        { double dt=Math.Min(seconds,step); RaceSimulation.Advance(game,dt);seconds-=dt;
          if(++guard>100000)throw new Exception("advance guard"); }
    }
    static void Tap(GameState game)
    { Check(RaceSimulation.Press(game,0)&&RaceSimulation.Press(game,1),"both real presses accepted"); }
    static void MoveTo(GameState game,double target,double step)
    {
        int guard=0;
        while(game.Phase=="racing"&&target-game.Players[0].Distance>2e-7)
        {
            // Public maximum horizontal speed, independent of production event predictors.
            double recovery=game.Players[0].Recovering;
            RaceSimulation.Advance(game,recovery>0 ? Math.Min(step,recovery)
                : Math.Min(step,Math.Max(1e-9,(target-game.Players[0].Distance)/440)));
            if(++guard>100000)throw new Exception(context+": distance guard x="+game.Players[0].Distance+" target="+target);
        }
        foreach(var player in game.Players)Near(player.Distance,target,"actual distance target",2e-6);
    }
    static GameState Prepare(Track track,FlightSettings flight,int side,double step)
    {
        var game=Start(track,flight);
        if(side==0)
        {
            Tap(game);
            if(flight.FirstAction==FirstAction.Flip) Advance(game,FlipTime(Span)+1e-9,step);
            else
            {

                    weighted=Math.Max(weighted,release.X)+release.Weight;
                }
                worstConservative=Math.Max(worstConservative,conservative);worstWeighted=Math.Max(worstWeighted,weighted);
            }
            Console.WriteLine(track.Id+" subsets="+(1<<releases.Count)+" latest="+releases[releases.Count-1].X.ToString("R")
                +" conservativeCompleteX="+worstConservative.ToString("R")+" weightedCompleteX="+worstWeighted.ToString("R")
                +" finish="+finish.ToString("R")+" conservativeMargin="+(finish-worstConservative).ToString("R"));
        }
    }
}

public static partial class TwoModesIndependentAudit
{
    static bool CompleteShiftedFull(Track track,FlightSettings flight,double[] route,double[] times,int changed,double shift)
    {
        var game=Start(track,flight);
        for(int i=0;i<route.Length&&game.Phase=="racing";i++)
        {
            if(i==changed)
            {double dt=times[i]+shift-game.Elapsed;if(dt<0)return false;Advance(game,dt,1d/144);}
            else
            {if(route[i]<game.Players[0].Distance-2e-6)return false;MoveTo(game,route[i],1d/144);}
            if(!RaceSimulation.Press(game,0)||!RaceSimulation.Press(game,1))return false;
        }
        Advance(game,100,1d/144);
        foreach(var p in game.Players)
            if(!p.FinishTime.HasValue||Math.Abs(p.FinishTime.Value-33.2)>5e-7||p.Collisions!=0||p.SlowsEarned!=0
                ||p.BoostsEarned!=9||p.CompletedBoosts!=9||p.FinishBoostsEarned!=1||p.ActiveEffect!=null||p.PendingEffects.Count!=0)return false;
        return Math.Abs(game.Elapsed-33.2)<5e-7;
    }
    static void FullRewardWindows()
    {
        int shiftedRuns=0;
        foreach(var flight in new[]{ExpectedMain,ExpectedFast})foreach(var source in TracksFull.All)
        {
            context="full window baseline "+source.Id+" "+flight.FirstAction;
            var track=TracksFull.Get(source.Id,flight);var route=TracksFull.FullRoute(source.Id,flight);
            double[] times;FullRouteGame(track,flight,1,1d/144,out times);
            double minEarly=1,minLate=1;
            for(int index=0;index<route.Length;index++)
            {
                double early=0,late=0;
                foreach(int direction in new[]{-1,1})
                {
                    for(int tick=1;tick<=60;tick++)
                    {
                        double shift=direction*tick*.005;
                        context="full window "+source.Id+" "+flight.FirstAction+" input="+index+" shift="+shift;
                        shiftedRuns++;
                        if(!CompleteShiftedFull(track,flight,route,times,index,shift))break;
                        if(direction<0)early=tick*.005;else late=tick*.005;
                    }
                }
                Console.WriteLine("WINDOW "+source.Id+" "+flight.FirstAction+" input="+index+" X="+route[index].ToString("R")+" actualTime="+times[index].ToString("R")+" early="+early+" late="+late+(index==route.Length-1?" final":""));
                if(index<route.Length-1){minEarly=Math.Min(minEarly,early);minLate=Math.Min(minLate,late);}
                Check(early>0&&late>0,"every full input has a measured nonzero neighborhood");
            }
            Console.WriteLine("ORDINARY WINDOW MIN "+source.Id+" "+flight.FirstAction+" early="+minEarly+" late="+minLate);
        }
        Console.WriteLine("PASS measured full-input windows: "+shiftedRuns+" shifted full-route simulations; 5ms grid, one input at a time, stop at first failed sample, cap300ms per side. Final target reported separately.");
    }
