using System;
using FlipRace.Core;
public static partial class FlightIndependentAudit
{
    static int checks, games;
    static string context;
    static readonly double[] Steps = { 1d / 30, 1d / 60, 1d / 144, .317, 2.5, 100 };
    static readonly AirRepeatLimit[] Repeats = { AirRepeatLimit.None, AirRepeatLimit.One, AirRepeatLimit.Unlimited };
    static void C(bool yes, string message) { checks++; if (!yes) throw new Exception(context + ": " + message); }
    static void N(double actual, double expected, string message, double tolerance = 3e-7)
    { C(!double.IsNaN(actual) && Math.Abs(actual - expected) <= tolerance, message + ": " + actual.ToString("R") + " != " + expected.ToString("R")); }
    static double Away(int side) { return side == 1 ? -1 : 1; }
    static double WallY(int side) { return side == 1 ? 226 : 14; }
    static double Travel(double distance, double speed, double acceleration)
    { return 2 * distance / (Math.Sqrt(speed * speed + 2 * acceleration * distance) + speed); }
    static GameState Start(FlightSettings f, Track track = null)
    {
        var g = RaceSimulation.CreateTrial(track ?? new Track(220000, Array.Empty<Obstacle>()), f);
        C(RaceSimulation.Press(g, 0) && RaceSimulation.Press(g, 1), "both ready");
        RaceSimulation.Advance(g, 3); C(g.Phase == "racing", "countdown completed"); games++; return g;
    }
    static void M(GameState g, double duration, double step)
    {
        C(duration >= -1e-9, "nonnegative requested elapsed"); int guard = 0;
        while (duration > 1e-13 && g.Phase == "racing")
        { double dt = Math.Min(duration, step); RaceSimulation.Advance(g, dt); duration -= dt; if (++guard > 100000) throw new Exception("advance guard"); }
    }
    static void Tap(GameState g) { C(RaceSimulation.Press(g, 0) && RaceSimulation.Press(g, 1), "both actual input edges accepted"); }
    // Mirror fixture: establish the ceiling by an actual no-repeat flip first.
    // Only the immutable tested settings reference changes at stationary support;
    // no position, velocity, support, charge, clock or result is fabricated.
    static GameState Prepare(FlightSettings f, int side, double step, Track track = null)
    {
        if (side == 1) return Start(f, track);
        var prep = new FlightSettings(FirstAction.Flip, f.Law, f.AirRepeats, f.GravityAcceleration,
            f.FlipImpulse, f.JumpImpulse, f.LinearSpeed, f.LinearJumpHeight);
        var g = Start(prep, track); Tap(g);
        M(g, f.Law == MotionLaw.Linear ? 212 / f.LinearSpeed : Travel(212, f.FlipImpulse, f.GravityAcceleration), step);
        foreach (var p in g.Players) C(p.Support == SupportKind.Wall && p.SupportSide == 0 && !p.Flipping, "real ceiling preparation");
        g.Flight = f; return g;
    }
    static void Support(GameState g, int side, AirRepeatLimit repeats)
    {
        foreach (var p in g.Players)
        {
            N(RaceSimulation.GetPlayerCenterY(p), WallY(side), "wall endpoint"); N(p.VerticalSpeed, 0, "wall velocity");
            C(!p.Flipping && !p.FlightActive && p.Support == SupportKind.Wall && p.SupportSide == side, "stable completed wall flight");
            C(p.AirFlipAvailable == (repeats != AirRepeatLimit.None), "recharge obeys independent limit");
        }
    }
    static void Measurement(PlayerState p, double duration, double height, double endedAt, int side, bool interrupted = false, string origin = null)
    {
        C(p.LastFlight != null, "measurement exists"); N(p.LastFlight.Duration, duration, "measured active flight time");
        N(p.LastFlight.MaxDistance, height, "analytic maximum distance"); N(p.LastFlight.EndedAt, endedAt, "physical end timestamp");
        C(p.LastFlight.Interrupted == interrupted && p.LastFlight.OriginSide == side && p.LastFlight.OriginObstacleId == origin, "measurement flags and origin");
    }
    static void FreeCycles()
    {
        foreach (var action in new[] { FirstAction.Flip, FirstAction.Jump })
        foreach (double gravity in new[] { 662.5, 2700 })
        foreach (double impulse in action == FirstAction.Flip ? new[] { 0d, 420 } : new[] { 200d, 550 })
        foreach (var repeat in Repeats) foreach (int side in new[] { 0, 1 }) foreach (double step in Steps)
        {
            context = "gravity " + action + " g" + gravity + " u" + impulse + " r" + repeat + " side" + side + " dt" + step;
            var f = new FlightSettings(action, MotionLaw.Gravity, repeat, gravity,
                action == FirstAction.Flip ? impulse : 70, action == FirstAction.Jump ? impulse : 400);
            var g = Prepare(f, side, step); double start = g.Elapsed; Tap(g); double sign = Away(side);
            double expectedTime, maximum; int endSide;
            if (action == FirstAction.Flip) { expectedTime = Travel(212, impulse, gravity); maximum = 212; endSide = 1 - side; }
            else
            {
                maximum = impulse * impulse / (2 * gravity); expectedTime = 2 * impulse / gravity; endSide = side;
                if (maximum > 212)
                {
                    double wall = (impulse - Math.Sqrt(impulse * impulse - 424 * gravity)) / gravity;
                    M(g, wall, step); foreach (var p in g.Players)
                    {
                        N(RaceSimulation.GetPlayerCenterY(p), WallY(1 - side), "opposite wall before apex");
                        C(p.Flipping && p.FlightActive && p.Support == SupportKind.None, "wrong wall neither ends flight nor supports");
                        N(p.FlightMaxDistance, 212, "actual maximum clipped by wall");
                    }
                    expectedTime = wall + Math.Sqrt(424 / gravity); maximum = 212;
                }
            }
            M(g, start + expectedTime - g.Elapsed + 1e-9, step); Support(g, endSide, repeat);
            foreach (var p in g.Players) Measurement(p, expectedTime, maximum, start + expectedTime, side);
        }
        foreach (var action in new[] { FirstAction.Flip, FirstAction.Jump })
        foreach (double fullTime in new[] { 1.2, .8, .4 })
        foreach (double height in action == FirstAction.Jump ? new[] { 48d, 112, 192 } : new[] { 112d })
        foreach (var repeat in Repeats) foreach (int side in new[] { 0, 1 }) foreach (double step in Steps)
        {
            context = "linear " + action + " T" + fullTime + " H" + height + " r" + repeat + " side" + side + " dt" + step;
            double speed = 212 / fullTime, duration = action == FirstAction.Flip ? fullTime : 2 * height / speed;
            var f = new FlightSettings(action, MotionLaw.Linear, repeat, 2700, 420, 550, speed, height);
            var g = Prepare(f, side, step); double start = g.Elapsed; Tap(g);
            if (action == FirstAction.Jump)
            {
                M(g, height / speed, step); foreach (var p in g.Players)
                {
                    N(RaceSimulation.GetPlayerCenterY(p), WallY(side) + Away(side) * height, "exact linear apex");
                    C(p.Flipping && p.FlightActive && p.Support == SupportKind.None, "apex is not support");
                    N(p.VerticalSpeed, -Away(side) * speed, "apex reverses without kick");
                }
            }
            M(g, start + duration - g.Elapsed + 1e-9, step); Support(g, action == FirstAction.Flip ? 1 - side : side, repeat);
            foreach (var p in g.Players) Measurement(p, duration, action == FirstAction.Flip ? 212 : height, start + duration, side);
        }
    }
    public static void Run()
    {
        checks = games = 0; FreeCycles(); RepeatRules(); StoneCases(); InterruptAndReset();
        Console.WriteLine("PASS " + checks + " independent FLIGHT assertions / " + games + " games:12 combinations, extreme mirrored cycles, analytic measurements, air repeats, stone contacts/walkoff, collision interruption/reset.");
    }
    public static void Main() { Run(); }
}
public static partial class FlightIndependentAudit
{
    static void RepeatRules()
    {
        foreach (var law in new[] { MotionLaw.Linear, MotionLaw.Gravity })
        foreach (var action in new[] { FirstAction.Flip, FirstAction.Jump })
        foreach (var repeat in Repeats) foreach (int side in new[] { 0, 1 })
        foreach (double age in action == FirstAction.Jump && law == MotionLaw.Gravity ? new[] { .12, .55, .95 } : new[] { .12, .55 }) foreach (double step in Steps)
        {
            context = "repeats " + law + "/" + action + "/" + repeat + " side" + side + " age" + age + " dt" + step;
            var f = new FlightSettings(action, law, repeat, 662.5, 0, 400, 265, 112);
            var g = Prepare(f, side, step); double start = g.Elapsed; Tap(g); M(g, age, step);
            foreach (var p in g.Players) C(p.Flipping, "tap is actually airborne");
            var p0 = g.Players[0]; double y = p0.Y, vy = p0.VerticalSpeed, time = p0.FlightDuration, max = p0.FlightMaxDistance;
            int gravity = p0.Gravity; bool accepted = RaceSimulation.Press(g, 0);
            C(accepted == (repeat != AirRepeatLimit.None), "first accepted count");
            N(p0.Y, y, "air tap no positional snap"); N(p0.FlightDuration, time, "air tap does not reset measurement time");
            N(p0.FlightMaxDistance, max, "air tap does not reset measured maximum");
            if (law == MotionLaw.Gravity || !accepted) N(p0.VerticalSpeed, vy, "gravity or rejected edge no impulse");
            else N(p0.VerticalSpeed, action == FirstAction.Jump ? Away(side) * f.LinearSpeed : -vy, "linear first action semantics");
            C(p0.Gravity == (accepted ? action == FirstAction.Jump && law == MotionLaw.Linear ? (int)Away(side) : -gravity : gravity), "gravity/target sign correct");
            C(g.Players[1].Gravity == gravity && g.Players[1].Y == y && g.Players[1].VerticalSpeed == vy, "other player unchanged");
            double firstVelocity = p0.VerticalSpeed; int firstGravity = p0.Gravity;
            C(RaceSimulation.Press(g, 0) == (repeat == AirRepeatLimit.Unlimited), "second separate edge count");
            if (repeat == AirRepeatLimit.Unlimited)
            {
                N(p0.VerticalSpeed, law == MotionLaw.Linear ? -firstVelocity : firstVelocity, "subsequent linear reversal vs inertia");
                C(p0.Gravity == -firstGravity, "second reversal");
                for (int i = 0; i < 32; i++) C(RaceSimulation.Press(g, 0), "unlimited zero-delay edges not throttled");
                N(p0.Y, y, "many edges still no teleport");
            }
            double expectedY = RaceSimulation.GetPlayerCenterY(p0), expectedV = p0.VerticalSpeed, a = law == MotionLaw.Gravity ? p0.Gravity * f.GravityAcceleration : 0;
            M(g, .01, step); N(RaceSimulation.GetPlayerCenterY(p0), expectedY + expectedV * .01 + .5 * a * .0001, "post-tap actual movement");
            N(p0.VerticalSpeed, expectedV + a * .01, "post-tap no kick");
        }
        // First edge exactly at or on either side of the automatic linear apex.
        foreach (int side in new[] { 0, 1 }) foreach (double offset in new[] { -1e-7, 0, 1e-7 }) foreach (double step in Steps)
        {
            context = "linear apex input side" + side + " offset" + offset + " dt" + step;
            var f = new FlightSettings(FirstAction.Jump, MotionLaw.Linear, AirRepeatLimit.One);
            var g = Prepare(f, side, step); double start = g.Elapsed; Tap(g); M(g, 112d / 265 + offset, step);
            double y = RaceSimulation.GetPlayerCenterY(g.Players[0]); Tap(g);
            foreach (var p in g.Players) { N(p.VerticalSpeed, Away(side) * 265, "first tap heads opposite original wall"); C(!p.LinearJumpActive && !p.AirFlipAvailable, "automatic apex cancelled and repeat spent"); }
            double rest = Math.Abs(WallY(1 - side) - y) / 265; M(g, rest + 1e-9, step); Support(g, 1 - side, AirRepeatLimit.One);
            foreach (var p in g.Players) Measurement(p, g.Elapsed - start - 1e-9, 212, g.Elapsed - 1e-9, side);
        }
        foreach (var law in new[] { MotionLaw.Linear, MotionLaw.Gravity })
        foreach (var action in new[] { FirstAction.Flip, FirstAction.Jump })
        foreach (var repeat in new[] { AirRepeatLimit.One, AirRepeatLimit.Unlimited })
        foreach (int side in new[] { 0, 1 }) foreach (double step in Steps)
        {
            context = "completed repeat measurement " + law + "/" + action + "/" + repeat + " side" + side + " dt" + step;
            var f = new FlightSettings(action, law, repeat, 662.5, 70, 400);
            var g = Prepare(f, side, step); double start = g.Elapsed;
            double age = action == FirstAction.Flip ? .1 : law == MotionLaw.Linear ? .55 : .2;
            Tap(g); M(g, age, step); Tap(g);
            double duration, height; int endSide;
            if (law == MotionLaw.Linear)
            {
                if (action == FirstAction.Flip) { duration = 2 * age; height = age * 265; endSide = side; }
                else { double q = 224 - age * 265; duration = age + (212 - q) / 265; height = 212; endSide = 1 - side; }
            }
            else if (action == FirstAction.Flip)
            {
                double q = 70 * age + .5 * 662.5 * age * age, v = 70 + 662.5 * age;
                duration = age + (v + Math.Sqrt(v * v + 2 * 662.5 * q)) / 662.5;
                height = q + v * v / (2 * 662.5); endSide = side;
            }
            else
            {
                double q = 400 * age - .5 * 662.5 * age * age, v = 400 - 662.5 * age;
                duration = age + Travel(212 - q, v, 662.5); height = 212; endSide = 1 - side;
            }
            M(g, duration - age + 1e-9, step); Support(g, endSide, repeat);
            foreach (var p in g.Players) Measurement(p, duration, height, start + duration, side);
        }
        HiddenValues();
    }
    static void HiddenValues()
    {
        foreach (var law in new[] { MotionLaw.Linear, MotionLaw.Gravity })
        foreach (var action in new[] { FirstAction.Flip, FirstAction.Jump }) foreach (int side in new[] { 0, 1 })
        {
            context = "inactive values " + law + "/" + action + " side" + side;
            var first = new FlightSettings(action, law, AirRepeatLimit.Unlimited, 662.5, 70, 400, 265, 112);
            var second = law == MotionLaw.Linear
                ? new FlightSettings(action, law, AirRepeatLimit.Unlimited, 2700, 420, 550, 265, 112)
                : new FlightSettings(action, law, AirRepeatLimit.Unlimited, 662.5,
                    action == FirstAction.Flip ? 70 : 420, action == FirstAction.Jump ? 400 : 550, 530, 192);
            var a = Prepare(first, side, .317); var b = Prepare(second, side, .317); Tap(a); Tap(b);
            for (int frame = 0; frame < 120; frame++)
            {
                M(a, .01, .01); M(b, .01, .01);
                if (frame == 7 || frame == 23) { Tap(a); Tap(b); }
                for (int i = 0; i < 2; i++)
                {
                    var x = a.Players[i]; var y = b.Players[i];
                    N(x.Y, y.Y, "inactive parameters do not change trajectory"); N(x.VerticalSpeed, y.VerticalSpeed, "inactive parameters do not change speed");
                    C(x.Gravity == y.Gravity && x.Support == y.Support && x.AirFlipAvailable == y.AirFlipAvailable, "inactive parameters do not change semantics");
                    N(x.FlightMaxDistance, y.FlightMaxDistance, "inactive parameters do not change measurement");
                }
            }
            C(first.FlipImpulse == 70 && second.LinearJumpHeight == (law == MotionLaw.Linear ? 112 : 192), "settings remain immutable physical values");
        }
    }

public static partial class FlightIndependentAudit
{
    static GameState OnStone(FlightSettings tested, int side, double step, double height = 48, bool wide = false)
    {
        var stone = new Obstacle { Id = "audit-stone", Kind = side == 1 ? "floor" : "ceiling", X = 3000, Width = wide ? 3000 : 300, Height = height };
        var track = new Track(220000, new[] { stone }) { RiskZones = new[] {
            new RiskZone { Id = "audit-pad", Kind = "landing", ObstacleId = stone.Id,
                X = wide ? 4500 : 3100, LandingWidth = wide ? 3000 : 52 } } };
        var prep = new FlightSettings(FirstAction.Flip, tested.Law, tested.AirRepeats, tested.GravityAcceleration,
            tested.FlipImpulse, tested.JumpImpulse, tested.LinearSpeed, tested.LinearJumpHeight);
        var g = Prepare(prep, 1 - side, step, track);
        double time = tested.Law == MotionLaw.Linear ? (212 - height) / tested.LinearSpeed
            : Travel(212 - height, tested.FlipImpulse, tested.GravityAcceleration);
        M(g, 3100d / 220 - time - g.Elapsed, step); Tap(g); M(g, time + 1e-9, step);
        foreach (var p in g.Players)
        {
            C(p.Support == SupportKind.Stone && p.SupportSide == side && p.SupportObstacleId == stone.Id && !p.Flipping, "actual stable marked-stone preparation");
            N(p.Distance, 3100, "real landing X", 1e-5); N(RaceSimulation.GetPlayerCenterY(p), WallY(side) + Away(side) * height, "real stone plane");
            C(p.Collisions == 0 && p.BoostsEarned == 1, "preparation is clean actual once-reward");
        }
        g.Flight = tested; return g;
    }
    static void StoneCases()
    {
        foreach (var law in new[] { MotionLaw.Linear, MotionLaw.Gravity })
        foreach (var action in new[] { FirstAction.Flip, FirstAction.Jump })
        foreach (var repeat in Repeats) foreach (int side in new[] { 0, 1 }) foreach (double step in Steps)
        {
            context = "stone " + law + "/" + action + "/" + repeat + " side" + side + " dt" + step;
            var f = new FlightSettings(action, law, repeat);
            var g = OnStone(f, side, step); double gap = 3300 - g.Players[0].Distance, boost = g.Players[0].BoostRemaining;
            double toEdge = gap <= boost * 330 ? gap / 330 : boost + (gap - boost * 330) / 220;
            double edgeAt = g.Elapsed + toEdge; M(g, toEdge + 1e-9, step);
            foreach (var p in g.Players)
            {
                C(p.Support == SupportKind.None && p.Flipping && p.FlightActive, "actual walkoff starts independent measured fall");
                C(!p.LinearJumpActive, "walkoff does not invent jump apex");
                N(p.VerticalSpeed, law == MotionLaw.Linear ? -Away(side) * f.LinearSpeed : -Away(side) * f.GravityAcceleration * 1e-9, "walkoff no departure impulse");
            }
            double fall = law == MotionLaw.Linear ? 48 / f.LinearSpeed : Math.Sqrt(96 / f.GravityAcceleration);
            M(g, fall, step); Support(g, side, repeat);
            foreach (var p in g.Players) { Measurement(p, fall, 48, edgeAt + fall, side, false, "audit-stone"); C(p.Collisions == 0, "no trailing-face phantom hit"); }

            var d = OnStone(f, side, step, 48, true); double start = d.Elapsed; Tap(d);
            double duration = law == MotionLaw.Linear ? (action == FirstAction.Flip ? 164 : 224) / f.LinearSpeed
                : action == FirstAction.Flip ? Travel(164, f.FlipImpulse, f.GravityAcceleration) : 2 * f.JumpImpulse / f.GravityAcceleration;
            double maximum = action == FirstAction.Flip ? 164 : law == MotionLaw.Linear ? 112 : f.JumpImpulse * f.JumpImpulse / (2 * f.GravityAcceleration);
            M(d, duration + 1e-9, step);
            foreach (var p in d.Players)
            {
                C(!p.Flipping && !p.FlightActive && p.Collisions == 0, "raised departure and return/arrival complete cleanly");
                C(p.Support == (action == FirstAction.Flip ? SupportKind.Wall : SupportKind.Stone), "actual reached support");
                C(p.AirFlipAvailable == (repeat != AirRepeatLimit.None) && p.BoostsEarned == 1, "support recharges repeats but not once reward");
                Measurement(p, duration, maximum, start + duration, side, false, "audit-stone");
            }
        }
        foreach (double stoneHeight in new[] { 20d, 48 }) foreach (var repeat in Repeats)
        foreach (double speed in new[] { 212 / 1.2, 265, 530 }) foreach (int side in new[] { 0, 1 }) foreach (double step in Steps)
        {
            context = "linear stone jump wall-before-or-at-apex H" + stoneHeight + " v" + speed + " r" + repeat + " side" + side + " dt" + step;
            var f = new FlightSettings(FirstAction.Jump, MotionLaw.Linear, repeat, 2700, 420, 550, speed, 192);
            var g = OnStone(f, side, step, stoneHeight, true); double start = g.Elapsed, height = 212 - stoneHeight, oneWay = height / speed; Tap(g);
            M(g, oneWay, step);
            foreach (var p in g.Players)
            {
                N(RaceSimulation.GetPlayerCenterY(p), WallY(1 - side), "wall precedes or exactly ties apex");
                N(p.VerticalSpeed, -Away(side) * speed, "single automatic reflection");
                C(p.Support == SupportKind.None && p.Flipping && p.FlightActive && p.LinearJumpReturning, "reflection is still one jump");
                C(p.AirFlipAvailable == (repeat != AirRepeatLimit.None), "wall reflection grants no new charge");
                N(p.FlightDuration, oneWay, "same measurement clock"); N(p.FlightMaxDistance, height, "max uses reachable height not selected192");
            }
            M(g, oneWay + 1e-9, step);
            foreach (var p in g.Players)
            {
                C(p.Support == SupportKind.Stone && !p.Flipping && p.Collisions == 0 && p.BoostsEarned == 1, "return is legal same marked stone once");
                Measurement(p, 2 * oneWay, height, start + 2 * oneWay, side, false, "audit-stone");
            }
        }
    }
    static void InterruptAndReset()
    {
        foreach (var law in new[] { MotionLaw.Linear, MotionLaw.Gravity })
        foreach (var action in new[] { FirstAction.Flip, FirstAction.Jump })
        foreach (var repeat in Repeats) foreach (int side in new[] { 0, 1 }) foreach (double step in Steps)
        {
            context = "interrupted " + law + "/" + action + "/" + repeat + " side" + side + " dt" + step;
            var f = new FlightSettings(action, law, repeat);
            var track = new Track(220000, new[] { new Obstacle { Id = "audit-blocker", Kind = "gate", X = 3000, Width = 20, GapTop = 240, GapBottom = 240 } });
            var g = Prepare(f, side, step, track); M(g, (2986d - 22) / 220 - g.Elapsed, step); double start = g.Elapsed; Tap(g); M(g, .1, step);
            double maximum = law == MotionLaw.Linear ? f.LinearSpeed * .1
                : (action == FirstAction.Jump ? f.JumpImpulse : f.FlipImpulse) * .1
                    + (action == FirstAction.Jump ? -1 : 1) * .5 * f.GravityAcceleration * .01;
            foreach (var p in g.Players)
            {
                C(p.Collisions == 1 && p.Recovering > 0 && !p.FlightActive, "first actual hit finalizes interrupted metric");
                Measurement(p, .1, maximum, start + .1, side, true);
                C(!RaceSimulation.Press(g, p.Id), "recovery input rejected");
            }
            var first = g.Players[0].LastFlight; M(g, 3, step);
            foreach (var p in g.Players) { C(!p.Flipping && p.LastFlight.Interrupted && !p.FlightActive, "later landing cannot fake a clean measurement"); Measurement(p, .1, maximum, start + .1, side, true); }
            C(ReferenceEquals(first, g.Players[0].LastFlight), "interrupted immutable sample preserved");
            Tap(g); M(g, 3, step); foreach (var p in g.Players) C(p.LastFlight != null && !p.LastFlight.Interrupted && !p.FlightActive, "new actual launch can produce next clean sample");
        }
        foreach (var f in new[] { FlightPresets.Moderate, FlightPresets.Linear, FlightPresets.Jump, FlightPresets.Unlimited })
        foreach (double step in Steps)
        {
            context = "incomplete finish/rematch";
            var g = Start(f, new Track(22, Array.Empty<Obstacle>())); Tap(g); M(g, .2, step);
            foreach (var p in g.Players) C(p.FinishTime.HasValue && p.LastFlight == null && !p.FlightActive, "unfinished flight at race finish is not completed measurement");
            C(RaceSimulation.Press(g, 0) && g.Phase == "finished", "rematch waits for both"); C(RaceSimulation.Press(g, 1) && g.Phase == "countdown", "second ready starts rematch");
            C(ReferenceEquals(g.Flight, f), "immutable actual settings retained across rematch");
            foreach (var p in g.Players) C(p.LastFlight == null && !p.FlightActive && !p.LinearJumpActive && p.Distance == 0, "rematch clears all measurement and jump dynamics");
        }
    }
}
